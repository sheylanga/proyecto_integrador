enable
configure terminal
hostname DIST-SW-01

! --- Hardening de Acceso y Servicios Criptográficos ---
no ip domain-lookup
ip domain-name dermacosmetics.tecnm.edu.mx
service password-encryption
enable secret 9 $9$vT1.Qe$k0r1mZ1yP4w9d38XzB8j.8L9eA1mPq!81283910

! Generación de llaves criptográficas RSA para SSH v2 seguro
crypto key generate rsa general-keys modulus 2048
ip ssh version 2
ip ssh time-out 60
ip ssh authentication-retries 3

! Creación de usuario de administración con máximo nivel de privilegio (15)
username netadmin privilege 15 secret 9 $9$AdminNet2026$8Xk9.1bV8sL02qW!

! Banner legal formal fundamentado en el Código Penal Federal Art. 211 bis
banner motd ^C
*******************************************************************************
*                        AVISO DE CONFIDENCIALIDAD LEGAL                      *

* CUALQUIER INTENTO DE INTRUSION O MODIFICACION SERA CONSIGNADO ANTE LA       *
* FISCALIA GENERAL DE LA REPUBLICA BAJO LOS ARTICULOS 211 BIS 1 AL 211 BIS 7  *

! --- Creación y Nomenclatura de VLANs Corporativas (802.1Q) ---

! --- Interfaces Virtuales de Conmutador (SVIs) y Redundancia HSRP (Activo) ---
! VLAN 10: Cluster Base de Datos (SQL Server & PostgreSQL)
interface Vlan10
 description SVI GATEWAY VIRTUAL CLUSTER BASE DE DATOS
 ip address 10.40.5.2 255.255.255.192

 standby 10 preempt
 standby 10 authentication md5 key-string PassBdHsrp2026
 no shutdown
exit

! --- Listas de Control de Acceso Extendidas (ACLs) para Aislamiento de BD ---
! Rúbrica Criterio 3.4: "Aislamiento de tráfico entre VLANs para proteger los servidores de base de datos"
ip access-list extended ACL_PROTECCION_CLUSTER_BD
 remark Regla 1: Permitir a servidores web de la DMZ (VLAN 20) conectar a SQL Server (TDS 1433)
 permit tcp 10.40.4.0 0.0.0.127 10.40.5.0 0.0.0.63 eq 1433
 remark Regla 2: Permitir a servidores web de la DMZ (VLAN 20) conectar a PostgreSQL (5432)
 permit tcp 10.40.4.0 0.0.0.127 10.40.5.0 0.0.0.63 eq 5432
 remark Regla 3: Permitir al host bastión de Admin TI (10.40.5.70) gestión SSH/RDP y SQL
 permit tcp host 10.40.5.70 10.40.5.0 0.0.0.63 eq 1433
 permit tcp host 10.40.5.70 10.40.5.0 0.0.0.63 eq 5432
 permit tcp host 10.40.5.70 10.40.5.0 0.0.0.63 eq 22
 permit tcp host 10.40.5.70 10.40.5.0 0.0.0.63 eq 3389
 remark Regla 4: Permitir ping ICMP unicamente desde la red de gestion (VLAN 99)
 permit icmp 10.40.5.96 0.0.0.31 10.40.5.0 0.0.0.63 echo
 permit icmp 10.40.5.96 0.0.0.31 10.40.5.0 0.0.0.63 echo-reply
 remark Regla 5: DENEGACION EXPLICITA Y AUDITORIA de cualquier otro acceso a la base de datos
 deny ip any 10.40.5.0 0.0.0.63 log
 remark Regla 6: Permitir el resto del trafico normal inter-VLAN
 permit ip any any
exit

! Aplicar la ACL entrante en la interfaz SVI de la VLAN de Base de Datos
interface Vlan10
 ip access-group ACL_PROTECCION_CLUSTER_BD out
exit

! --- Configuración de Enrutamiento Dinámico OSPFv2 (Área 0 Backbone) ---
router ospf 1
 router-id 1.1.1.1
 log-adjacency-changes
 passive-interface default
 no passive-interface GigabitEthernet0/1
 no passive-interface GigabitEthernet0/2
 network 10.40.0.0 0.0.255.255 area 0
exit

! --- Configuración de Troncales 802.1Q y Agregación LACP ---
interface range GigabitEthernet0/1 - 2
 description ENLACES REDUNDANTES LACP HACIA CORE-SW-01
 channel-group 1 mode active
 no shutdown
exit

interface Port-channel 1
 description PORT-CHANNEL TRUNK 802.1Q HACIA CORE
 switchport trunk encapsulation dot1q
 switchport mode trunk
 switchport trunk allowed vlan 10,20,30,40,50,60,70,99
 switchport trunk native vlan 99
 no shutdown
exit
